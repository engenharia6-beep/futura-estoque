// ── Helpers básicos ───────────────────────────────────────────────────────────
function _ss()  { return SpreadsheetApp.openById(SPREADSHEET_ID); }
function _sh(n) { return _ss().getSheetByName(n); }

function _norm(s) {
  return String(s || '').trim()
    .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
    .toLowerCase()
    .replace(/[^a-z0-9 ]/g, '')
    .replace(/\s+/g, ' ')
    .trim();
}

function _idxCol(header, nome) {
  const alvo = _norm(nome);
  return header.findIndex(v => _norm(v) === alvo);
}

function _montarLinhaMovimento(header, campos) {
  const linha = new Array(header.length).fill('');
  const mapa  = {
    'id':          campos.id,
    'qrcode':      campos.qrcode      || '',
    'data':        campos.data,
    'tipo':        campos.tipo        || '',
    'n da op':     campos.op          || '',
    'op':          campos.op          || '',
    'codigo':      campos.codigo      || '',
    'descricao':   campos.descricao   || '',
    'qtde':        campos.qtde        ?? 0,
    'obs':         campos.obs         || '',
    'foto':        '',
    'endereco':    campos.endereco    || '',
    'user':        campos.usuario     || '',
    'usuario':     campos.usuario     || '',
    'computer':    campos.computer    || '',
    'dispositivo': campos.computer    || '',
    'localizacao': campos.localizacao || '',
    'chave':       campos.chave       || '',
    'ip':          campos.ip          || '',
  };
  const contagem = {};
  header.forEach((colName, i) => {
    const key = _norm(colName);
    if (mapa[key] === undefined) return;
    contagem[key] = (contagem[key] || 0) + 1;
    if (key === 'data') {
      const d = campos.data instanceof Date ? campos.data : new Date();
      linha[i] = contagem[key] === 1 ? d : new Date(d.getFullYear(), d.getMonth(), d.getDate());
    } else {
      linha[i] = mapa[key] !== undefined && mapa[key] !== null ? mapa[key] : '';
    }
  });
  return linha;
}

// ── Helpers compartilhados ───────────────────────────────────────────────────
// Endereço é 1 só por item (fixo no Cadastro/Cadastro_PA, não varia por
// movimento) — aqui só somamos o saldo total por código para validação
// dentro de um lote, sem quebrar por endereço.
function _saldosEmMemoria(data, header) {
  const idxN = {};
  header.forEach((v, i) => { const k = _norm(v); if (!idxN[k]) idxN[k] = i; });
  const iT = idxN['tipo']   ?? -1;
  const iC = idxN['codigo'] ?? -1;
  const iQ = idxN['qtde']   ?? -1;
  const mapa = {};
  for (let i = 1; i < data.length; i++) {
    const r   = data[i];
    const cod = String(r[iC] || '').trim();
    if (!cod) continue;
    const t   = String(r[iT] || '').toUpperCase();
    const q   = Number(r[iQ] || 0);
    const inc = t === 'ENTRADA' ? q : -q;
    mapa[cod] = (mapa[cod] || 0) + inc;
  }
  return mapa;
}

// Lê o endereço fixo (coluna D) de um ou mais códigos do Cadastro.
function _mapaEnderecoCadastro() {
  const sh = _sh(NOME_ABA_CADASTRO);
  if (!sh) return {};
  const data = sh.getDataRange().getValues();
  const h    = data[0];
  const iCOD = _idxCol(h, 'CÓDIGO');
  const iEND = _idxCol(h, 'Endereço');
  const mapa = {};
  if (iCOD < 0 || iEND < 0) return mapa;
  for (let i = 1; i < data.length; i++) {
    const cod = String(data[i][iCOD] || '').trim();
    if (cod) mapa[cod] = String(data[i][iEND] || '').trim();
  }
  return mapa;
}

// ── Cache de listas (CacheService) ──────────────────────────────────────────
// CacheService tem limite de 100KB POR CHAVE. Listas de Insumos/PA
// (976/446 itens) passam disso facilmente, então guardamos em pedaços
// (chunks) de string e remontamos na leitura. TTL curto (segundos) —
// não invalidamos no write, só aceitamos que a lista pode ficar até
// TTL segundos desatualizada após alguém editar algo. obterSaldo/
// obterSaldoPA (usados nas validações de gravação) NÃO usam esse cache
// — continuam sempre lendo a planilha na hora.
const _CACHE_CHUNK = 40000; // chars por chave, com margem sobre o limite de 100KB
function _cacheGetLista(key) {
  try {
    const cache = CacheService.getScriptCache();
    const n = Number(cache.get(key + '_n') || 0);
    if (!n) return null;
    const chaves = [];
    for (let i = 0; i < n; i++) chaves.push(key + '_' + i);
    const partes = cache.getAll(chaves);
    let json = '';
    for (let i = 0; i < n; i++) {
      const p = partes[key + '_' + i];
      if (p == null) return null; // pedaço faltando/expirado — trata como cache miss
      json += p;
    }
    return JSON.parse(json);
  } catch(e) { return null; }
}
function _cacheSetLista(key, valor, ttlSegundos) {
  try {
    const json = JSON.stringify(valor);
    const n = Math.max(1, Math.ceil(json.length / _CACHE_CHUNK));
    const mapa = {};
    for (let i = 0; i < n; i++) mapa[key + '_' + i] = json.substring(i * _CACHE_CHUNK, (i + 1) * _CACHE_CHUNK);
    mapa[key + '_n'] = String(n);
    CacheService.getScriptCache().putAll(mapa, ttlSegundos);
  } catch(e) {} // se a gravação do cache falhar, segue sem cache — não é crítico
}

function _verificarGravacao(sh, ids, linhasAntes) {
  try {
    SpreadsheetApp.flush();
    if (linhasAntes == null) return { ok: true, faltando: [] };
    const linhasDepois = sh.getLastRow();
    const esperado     = linhasAntes + ids.length;
    if (linhasDepois < esperado)
      return { ok: false, faltando: ids, erro: `Esperado ${esperado} linhas, encontrado ${linhasDepois}.` };
    return { ok: true, faltando: [] };
  } catch(e) { return { ok: false, faltando: ids, erro: e.message }; }
}

// ════════════════════════════════════════════════════════════════════════════
//  doPost — roteador central
//  Recebe: { acao: string, dados: object }
//  Retorna: JSON com resultado da função correspondente
// ════════════════════════════════════════════════════════════════════════════
function doPost(e) {
  try {
    const { acao, dados } = JSON.parse(e.postData.contents);

    const rotas = {
      // ── Auth ──────────────────────────────────────────────────────────────
      'autenticar':             () => autenticarUsuario(dados.usuario, dados.senha),
      'alterarSenha':           () => alterarSenha(dados.nomeUsuario, dados.senhaAtual, dados.senhaNova),

      // ── Cadastro Insumos ──────────────────────────────────────────────────
      'listarCadastro':         () => listarItensCadastro(dados.filtroSituacao),
      'salvarCadastro':         () => salvarItemCadastro(dados),
      'obterFormulasInsumo':    () => obterCamposComFormulaCadastro(dados.codigo),
      'obterResumoDashboard':   () => obterResumoDashboard(),

      // ── Cadastro PA ───────────────────────────────────────────────────────
      'listarCadastroPA':       () => listarItensCadastroPA(dados.filtroSituacao),
      'salvarCadastroPA':       () => salvarItemCadastroPA(dados),
      'obterFormulasPA':        () => obterCamposComFormulaCadastroPA(dados.codigo),

      // ── Movimentos Insumos ────────────────────────────────────────────────
      'listarMovimentos':       () => listarMovimentos(),
      'gravarMovimento':        () => gravarMovimento(dados),
      'gravarMovimentosEmLote': () => gravarMovimentosEmLote(dados.movimentos),
      'obterSaldo':             () => ({ saldo: obterSaldo(dados.codigo) }),
      'obterEnderecosSaldo':    () => obterEnderecosSaldo(dados.codigo),

      // ── Movimentos PA ─────────────────────────────────────────────────────
      'listarMovimentosPA':     () => listarMovimentosPA(),
      'gravarMovimentoPA':      () => gravarMovimentoPA(dados),
      'gravarMovimentosEmLotePA': () => gravarMovimentosEmLotePA(dados.movimentos),
      'obterSaldoPA':           () => ({ saldo: obterSaldoPA(dados.codigo) }),
      'obterEnderecosSaldoPA':  () => obterEnderecosSaldoPA(dados.codigo),

      // ── OPs ───────────────────────────────────────────────────────────────
      'listarOPS':              () => listarOPS(),
      'prepararBOM':            () => prepararEListarBOM(dados.op),
      'prepararBOMComSaldo':    () => prepararBOMComSaldo(dados),
      'baixarBOMComTipo':       () => baixarBOMComTipo(dados.movimentos, dados.dadosBase),

      // ── Triangular (nova versão com baixa de insumos) ─────────────────────
      'pagarTriangular':        () => pagarOPTriangularPA(dados),

      // ── Mudança de endereço ───────────────────────────────────────────────
      'mudarEndereco':          () => mudarEndereco(dados),
      'mudarEnderecoPA':        () => mudarEnderecoPA(dados),

      // ── Embarque / DI ─────────────────────────────────────────────────────
      'listarEmbarque':         () => listarEmbarque(),
      'gravarEntradaDI':        () => gravarEntradaDI(dados),

      // ── Impressão ─────────────────────────────────────────────────────────
      'imprimirZPL':            () => imprimirZPL(dados.op, dados.zpl, dados.impressora),
      'imprimirZPLCadastro':    () => imprimirZPLCadastro(dados.codigo),
      'imprimirZPLCadastroPA':  () => imprimirZPLCadastroPA(dados.codigo),

function _resposta(dados) {

//  ESTRUTURA DE PRODUTOS — listar insumos de um PA
// ════════════════════════════════════════════════════════════════════════════
function listarEstruturaProduto(codigoPA) {
  try {
    const sh = _sh(NOME_ABA_ESTRUTURA);
    if (!sh) return { ok: false, mensagem: 'Aba "Estrutura de Produtos" não encontrada.' };
    const data   = sh.getDataRange().getValues();
    if (data.length < 2) return { ok: true, itens: [] };
    const header = data[0];

    const iCodPA   = _idxCol(header, 'CODIGO_PRODUTO');
    const iCodIns  = _idxCol(header, 'CODIGO_INSUMO');
    const iDescIns = _idxCol(header, 'DESCRICAO_INSUMO');
    const iQtde    = _idxCol(header, 'QTDE_UTILIZADA');
    const iOrdem   = _idxCol(header, 'ORDEM');

    if (iCodPA < 0 || iCodIns < 0 || iQtde < 0)
      return { ok: false, mensagem: 'Colunas obrigatórias não encontradas na Estrutura de Produtos.' };

    const cod = String(codigoPA || '').trim().toUpperCase();
    const itens = data.slice(1)
      .filter(r => String(r[iCodPA] || '').trim().toUpperCase() === cod &&
                   String(r[iCodIns] || '').trim() !== '')
      .map(r => ({
        codigoInsumo:  String(r[iCodIns]  || '').trim(),
        descricao:     iDescIns >= 0 ? String(r[iDescIns] || '').trim() : '',
        qtdeUtilizada: Number(r[iQtde] || 0),
        ordem:         iOrdem >= 0 ? String(r[iOrdem] || '').trim() : '',
      }));

    return { ok: true, itens };
  } catch(e) { return { ok: false, mensagem: 'Erro ao listar estrutura: ' + e.message }; }
}

// ════════════════════════════════════════════════════════════════════════════
//  _baixarInsumosDaEstrutura — baixa automática dos insumos ao pagar PA
//  Chamada internamente por gravarMovimentoPA e pagarOPTriangularPA
//
//  Lógica:
//  - Busca os insumos do PA na aba "Estrutura de Produtos"
//  - Calcula qtde = QTDE_UTILIZADA × qtdePABaixado
//  - Chama gravarMovimentosEmLote para cada insumo
//  - NÃO bloqueia: se um insumo tiver saldo insuficiente, registra aviso e continua
//
//  Retorna: { gravados: N, avisos: ['MSG1', 'MSG2'] }
// ════════════════════════════════════════════════════════════════════════════
function _baixarInsumosDaEstrutura(codigoPA, qtdePABaixado, dadosBase) {
  const resultado = { gravados: 0, avisos: [] };
  try {
    const estrutura = listarEstruturaProduto(codigoPA);
    if (!estrutura.ok || !estrutura.itens.length) {
      resultado.avisos.push(`Nenhum insumo encontrado na Estrutura de Produtos para ${codigoPA}.`);
      return resultado;
    }

    const _dadosMov   = _sh(NOME_ABA_MOVIMENTO).getDataRange().getValues();
    const saldosInsumo = _saldosEmMemoria(_dadosMov, _dadosMov[0]);
    const saldoTotal = (cod) => saldosInsumo[cod] || 0;

    const paraGravar = [];
    const avisos     = [];

    for (const item of estrutura.itens) {
      const qtdeNecessaria = item.qtdeUtilizada * qtdePABaixado;
      if (qtdeNecessaria <= 0) continue;

      const saldo = saldoTotal(item.codigoInsumo);
      if (saldo < qtdeNecessaria) {
        // Avisa mas não bloqueia — baixa o que tiver
        avisos.push(`Saldo insuficiente para ${item.codigoInsumo}: necessário ${qtdeNecessaria}, disponível ${saldo}.`);
        if (saldo <= 0) continue; // sem saldo nenhum, pula
      }

      paraGravar.push({
        codigo:    item.codigoInsumo,
        descricao: item.descricao,
        qtde:      Math.min(qtdeNecessaria, saldo), // baixa o que tiver
        tipo:      'SAIDA',
        op:        dadosBase.op        || '',
        usuario:   dadosBase.usuario   || '',
        computer:  dadosBase.computer  || '',
        localizacao: dadosBase.localizacao || '',
        ip:        dadosBase.ip        || '',
        obs:       `Baixa automática via PA ${codigoPA} — OP: ${dadosBase.op || ''}`,
        qrcode:    '',
        chave:     '',
        endereco:  '',
      });
    }

    if (paraGravar.length > 0) {
      const res = gravarMovimentosEmLote(paraGravar);
      if (res.ok) {
        resultado.gravados = res.gravados;
      } else {
        avisos.push('Erro ao gravar insumos: ' + res.mensagem);
      }
    }

    resultado.avisos = avisos;
  } catch(e) {
    resultado.avisos.push('Erro ao baixar insumos da estrutura: ' + e.message);
  }
  return resultado;
}

//  AUTENTICAÇÃO
// ════════════════════════════════════════════════════════════════════════════
function autenticarUsuario(usuario, senha) {
  try {
    const sh   = _sh(NOME_ABA_USUARIOS);
    if (!sh) return { ok: false, mensagem: 'Aba USUARIOS não encontrada.' };
    const data = sh.getDataRange().getValues();
    let hRow   = -1;
    for (let i = 0; i < data.length; i++) {
      if (data[i].some(c => String(c).trim() === 'Usuario')) { hRow = i; break; }
    }
    if (hRow === -1) return { ok: false, mensagem: 'Cabeçalho não encontrado.' };
    const h   = data[hRow];
    const idx = {};
    h.forEach((v, i) => { idx[String(v).trim()] = i; });
    const row = data.slice(hRow + 1)
      .filter(r => String(r[idx['Usuario']] || '').trim() !== '')
      .find(r =>
        String(r[idx['Usuario']] || '').toLowerCase().trim() === usuario.toLowerCase().trim() &&
        String(r[idx['Senha']]   || '').trim() === senha.trim()
      );
    if (!row) return { ok: false, mensagem: 'Usuário ou senha inválidos.' };
    if (String(row[idx['Ativo']] || '').toUpperCase().trim() !== 'SIM')
      return { ok: false, mensagem: 'Usuário inativo.' };
    return { ok: true, usuario: {
      id:      String(row[idx['ID']]      || ''),
      usuario: String(row[idx['Usuario']] || ''),
      nome:    String(row[idx['Nome']]    || ''),
      perfil:  String(row[idx['Perfil']]  || ''),
      empresa: String(row[idx['Empresa']] || ''),
      ativo:   String(row[idx['Ativo']]   || ''),
    }};
  } catch(e) { return { ok: false, mensagem: 'Erro: ' + e.message }; }
}

function alterarSenha(nomeUsuario, senhaAtual, senhaNova) {
  try {
    const sh   = _sh(NOME_ABA_USUARIOS);
    if (!sh) return { ok: false, mensagem: 'Aba USUARIOS não encontrada.' };
    const data   = sh.getDataRange().getValues();
    let hRow     = -1;
    for (let i = 0; i < data.length; i++) {
      if (data[i].some(c => String(c).trim() === 'Usuario')) { hRow = i; break; }
    }
    if (hRow === -1) return { ok: false, mensagem: 'Cabeçalho não encontrado.' };
    const header = data[hRow];
    const idx    = {};
    header.forEach((v, i) => { idx[String(v).trim()] = i; });
    for (let i = hRow + 1; i < data.length; i++) {
      const r = data[i];
      if (String(r[idx['Nome']] || '').trim() !== nomeUsuario) continue;
      if (String(r[idx['Ativo']] || '').toUpperCase().trim() !== 'SIM')
        return { ok: false, mensagem: 'Usuário inativo.' };
      if (String(r[idx['Senha']] || '').trim() !== senhaAtual)
        return { ok: false, mensagem: 'Senha atual incorreta.' };
      sh.getRange(i + 1, idx['Senha'] + 1).setValue(senhaNova);
      return { ok: true };
    }
    return { ok: false, mensagem: 'Usuário não encontrado.' };
  } catch(e) { return { ok: false, mensagem: 'Erro: ' + e.message }; }
}

function salvarItemCadastro(dados) {
  try {
    const sh     = _sh(NOME_ABA_CADASTRO);
    if (!sh) return { ok: false, mensagem: 'Aba Cadastro não encontrada.' };
    const data   = sh.getDataRange().getValues();
    const header = data[0];
    const iCOD   = _idxCol(header, 'CÓDIGO');
    if (iCOD < 0) return { ok: false, mensagem: 'Coluna CÓDIGO não encontrada.' };
    const codigo = String(dados.codigo || '').trim().toUpperCase();
    if (!codigo) return { ok: false, mensagem: 'Código obrigatório.' };
    let rowNum = -1;
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][iCOD] || '').trim().toUpperCase() === codigo) { rowNum = i + 1; break; }
    }
    if (rowNum > 0) {
      const formulas   = sh.getRange(rowNum, 1, 1, header.length).getFormulas()[0];
      const linhaAtual = [...data[rowNum - 1]];
      for (const [field, colNames] of Object.entries(_CAMPOS_CAD)) {
        if (!(field in dados)) continue;
        for (const col of colNames) {
          const ci = _idxCol(header, col);
          if (ci < 0) continue;
          if (formulas[ci]) continue;
          linhaAtual[ci] = _NUM_CAMPOS.includes(field) ? Number(dados[field] || 0) : (dados[field] ?? '');
          break;
        }
      }
      sh.getRange(rowNum, 1, 1, linhaAtual.length).setValues([linhaAtual]);
      return { ok: true, acao: 'atualizado', mensagem: 'Item atualizado!' };
    }
    if (!dados.descricao) return { ok: false, mensagem: 'Descrição obrigatória.' };
    sh.appendRow(header.map(col => {
      const c = String(col).trim();
      switch(c) {
        case 'CÓDIGO':     return codigo;
        case 'DESCRIÇÃO':  return dados.descricao || '';
        case 'Situação':   return dados.situacao  || 'ATIVO';
        default:           return '';
      }
    }));
    return { ok: true, acao: 'criado', mensagem: 'Item criado!' };
  } catch(e) { return { ok: false, mensagem: 'Erro: ' + e.message }; }
}

//  DASHBOARD — resumo leve (só contagens, sem montar/enviar os itens
//  completos). Antes o dashboard chamava listarCadastro + listarCadastroPA
//  inteiros (todas as colunas de ~1400 itens) só pra contar 3 números.
// ════════════════════════════════════════════════════════════════════════════
function obterResumoDashboard() {
  try {
    const shIns = _sh(NOME_ABA_CADASTRO);
    let totalInsumos = 0, criticos = 0;
    if (shIns) {
      const data = shIns.getDataRange().getValues();
      const h    = data[0];
      const iCOD = _idxCol(h, 'CÓDIGO');
      const iSIT = _idxCol(h, 'Situação');
      const iEA  = _idxCol(h, 'ESTOQUE_ATUAL');
      const iMIN = _idxCol(h, 'Estoque_Minimo');
      for (let i = 1; i < data.length; i++) {
        const r   = data[i];
        const cod = iCOD >= 0 ? String(r[iCOD]||'').trim() : '';
        if (!cod) continue;
        const sit = iSIT >= 0 ? String(r[iSIT]||'').trim().toUpperCase() : '';
        if (sit !== 'ATIVO' && sit !== '') continue; // mesmo filtro padrão de listarItensCadastro('ATIVO')
        totalInsumos++;
        const atual = iEA  >= 0 ? Number(r[iEA]||0)  : 0;
        const min   = iMIN >= 0 ? Number(r[iMIN]||0) : 0;
        if (min > 0 && atual <= min) criticos++;
      }
    }

    const shPA = _sh(NOME_ABA_CADASTRO_PA);
    let totalPA = 0;
    if (shPA) {
      const data = shPA.getDataRange().getValues();
      const h    = data[0];
      const iCOD = _idxCol(h, 'CÓDIGO') >= 0 ? _idxCol(h, 'CÓDIGO') : _idxCol(h, 'CODIGO');
      const iSIT = _idxCol(h, 'Situação');
      for (let i = 1; i < data.length; i++) {
        const r   = data[i];
        const cod = iCOD >= 0 ? String(r[iCOD]||'').trim() : '';
        if (!cod) continue;
        const sit = iSIT >= 0 ? String(r[iSIT]||'').trim().toUpperCase() : '';
        if (sit !== 'ATIVO' && sit !== '') continue;
        totalPA++;
      }
    }

    return { ok: true, totalInsumos, totalPA, criticos };
  } catch(e) { return { ok: false, mensagem: 'Erro: ' + e.message }; }
}

//  MOVIMENTOS INSUMOS
// ════════════════════════════════════════════════════════════════════════════
function listarMovimentos() {
  try {
    const sh = _sh(NOME_ABA_MOVIMENTO);
    if (!sh) return { ok: false, mensagem: 'Aba Movimento não encontrada.' };
    const data = sh.getDataRange().getValues();
    if (data.length < 2) return [];
    const h    = data[0];
    const idxN = {};
    h.forEach((v, i) => { const k = _norm(v); if (!idxN[k]) idxN[k] = i; });
    const celN = (r, ...keys) => {
      for (const k of keys) { if (idxN[k] != null) return String(r[idxN[k]] || ''); }
      return '';
    };
    const iD = h.map((v,i) => _norm(v) === 'data' ? i : -1).filter(i => i >= 0)[0] ?? -1;
    const fmt = (r, i) => i >= 0 && r[i]
      ? Utilities.formatDate(new Date(r[i]), Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm') : '';
    return data.slice(1).filter(r => celN(r,'id') !== '').map(r => ({
      id:        celN(r,'id'),
      data:      fmt(r, iD),
      tipo:      celN(r,'tipo'),
      op:        celN(r,'n da op','op'),
      codigo:    celN(r,'codigo'),
      descricao: celN(r,'descricao'),
      qtde:      Number(idxN['qtde'] != null ? r[idxN['qtde']] || 0 : 0),
      endereco:  celN(r,'endereco'),
      obs:       celN(r,'obs'),
      usuario:   celN(r,'user','usuario'),
    }));
  } catch(e) { return { ok: false, mensagem: 'Erro: ' + e.message }; }
}

// Lê o saldo já calculado na coluna ESTOQUE_ATUAL do Cadastro (coluna O), em
// vez de somar o histórico inteiro de Movimento — Cadastro tem 1 linha por
// item (tamanho fixo), Movimento cresce para sempre a cada transação.
function obterSaldo(codigo) {
  try {
    const sh   = _sh(NOME_ABA_CADASTRO);
    if (!sh) return 0;
    const data = sh.getDataRange().getValues();
    const h    = data[0];
    const iCOD = _idxCol(h, 'CÓDIGO');
    const iEA  = _idxCol(h, 'ESTOQUE_ATUAL');
    if (iCOD < 0 || iEA < 0) return 0;
    const row = data.slice(1).find(r => String(r[iCOD]||'').trim() === String(codigo).trim());
    return row ? Number(row[iEA]||0) : 0;
  } catch(e) { return 0; }
}

// Endereço é fixo por item (Cadastro, coluna D) — não varia por movimento.
// Retorna { saldo, enderecos:[{endereco,saldo}] } lendo só o Cadastro, sem
// tocar em Movimento. Mantém o mesmo formato de resposta de antes (lista de
// 1 endereço no máximo) para não exigir mudança no frontend.
function obterEnderecosSaldo(codigo) {
  try {
    const sh = _sh(NOME_ABA_CADASTRO);
    if (!sh) return { saldo: 0, enderecos: [] };
    const data = sh.getDataRange().getValues();
    const h    = data[0];
    const iCOD = _idxCol(h, 'CÓDIGO');
    const iEND = _idxCol(h, 'Endereço');
    const iEA  = _idxCol(h, 'ESTOQUE_ATUAL');
    if (iCOD < 0) return { saldo: 0, enderecos: [] };
    const row = data.slice(1).find(r => String(r[iCOD]||'').trim() === String(codigo).trim());
    if (!row) return { saldo: 0, enderecos: [] };
    const saldo    = iEA  >= 0 ? Number(row[iEA]||0) : 0;
    const endereco = iEND >= 0 ? String(row[iEND]||'').trim() : '';
    return { saldo, enderecos: endereco ? [{ endereco, saldo }] : [] };
  } catch(e) { return { saldo: 0, enderecos: [] }; }
}

function gravarMovimento(dados) {
  try {
    const codigo = String(dados.codigo || '').trim();
    if (!codigo) return { ok: false, mensagem: 'Código não informado.' };
    // 1 leitura só do Cadastro: situação + endereço + saldo (antes eram
    // 3 leituras separadas — _mapaStatusCadastro + _mapaEnderecoCadastro + obterSaldo)
    const infoCad = _mapaCadastroInfo(NOME_ABA_CADASTRO);
    const itemCad = infoCad[codigo];
    if (itemCad === undefined) return { ok: false, mensagem: `Item '${codigo}' não encontrado.` };
    if (itemCad.situacao === 'INATIVO')  return { ok: false, mensagem: `Item '${codigo}' está INATIVO.` };
    const sh = _sh(NOME_ABA_MOVIMENTO);
    if (!sh) return { ok: false, mensagem: 'Aba Movimento não encontrada.' };
    const dadosSh     = sh.getDataRange().getValues();
    const header      = dadosSh[0];
    const linhasAntes = dadosSh.length;
    // Idempotência: rejeita gravação duplicada se a chave já existe
    const chaveAtual = String(dados.chave || dados.chaveIdem || '').trim();
    if (chaveAtual) {
      const iChave = header.findIndex(v => _norm(v) === 'chave');
      if (iChave >= 0 && dadosSh.slice(1).some(r => String(r[iChave]||'').trim() === chaveAtual))
        return { ok: true, gravados: 0, jaGravado: true };
    }
    const isSaida     = String(dados.tipo||'').toUpperCase() === 'SAIDA';
    const qtde        = Number(dados.qtde || 0);
    const agora       = new Date();
    const endereco    = itemCad.endereco || '';

    // Checa duplicidade de negócio — mesmo código + tipo + qtde gravado nos últimos 90s.
    // Protege contra duplo-clique no botão Confirmar antes do modal fechar (a chaveIdem muda
    // a cada clique, e o OP ADJ- só tem granularidade de minuto — pode nem bater entre os cliques).
    {
      const iCodCol  = _idxCol(header, 'codigo');
      const iTipoCol = _idxCol(header, 'tipo');
      const iQtdeCol = _idxCol(header, 'qtde');
      const iDataCol = _idxCol(header, 'data');
      if (iCodCol >= 0 && iTipoCol >= 0 && iQtdeCol >= 0 && iDataCol >= 0) {
        const tipoAlvo = isSaida ? 'SAIDA' : 'ENTRADA';
        const jaGravado = dadosSh.slice(1).some(r => {
          if (String(r[iCodCol]||'').trim() !== codigo) return false;
          if (String(r[iTipoCol]||'').trim().toUpperCase() !== tipoAlvo) return false;
          if (Number(r[iQtdeCol]||0) !== qtde) return false;
          const t = new Date(r[iDataCol]).getTime();
          return !isNaN(t) && Math.abs(agora.getTime() - t) <= 90000;
        });
        if (jaGravado) return { ok: true, gravados: 0, jaGravado: true };
      }
    }

    if (isSaida) {
      const saldoTotal = itemCad.saldo;
      if (qtde > saldoTotal) return { ok: false, mensagem: `Saldo insuficiente para '${codigo}'. Disponível: ${saldoTotal}.` };
    }
    const id  = Utilities.getUuid().substring(0,8);
    const ids = [id];
    sh.appendRow(_montarLinhaMovimento(header, Object.assign({}, dados, { id, data: agora, qtde, endereco, codigo })));
    const verif = _verificarGravacao(sh, ids, linhasAntes);
    if (!verif.ok) return { ok: false, mensagem: 'Movimento não confirmado. ' + (verif.erro||'') };
    return { ok: true, gravados: ids.length };
  } catch(e) { return { ok: false, mensagem: 'Erro: ' + e.message }; }
}

function listarMovimentosPA() {
  try {
    const sh = _sh(NOME_ABA_MOVIMENTO_PA);
    if (!sh) return { ok: false, mensagem: 'Aba Movimento_PA não encontrada.' };
    const data = sh.getDataRange().getValues();
    if (data.length < 2) return [];
    const h    = data[0];
    const idxN = {};
    h.forEach((v, i) => { const k = _norm(v); if (!idxN[k]) idxN[k] = i; });
    const celN = (r, ...keys) => { for (const k of keys) { if (idxN[k] != null) return String(r[idxN[k]]||''); } return ''; };
    const iD = idxN['data'] ?? -1;
    const fmt = (r, i) => i >= 0 && r[i] ? Utilities.formatDate(new Date(r[i]), Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm') : '';
    return data.slice(1).filter(r => celN(r,'id') !== '').map(r => ({
      id: celN(r,'id'), data: fmt(r,iD), tipo: celN(r,'tipo'), op: celN(r,'op','n da op'),
      codigo: celN(r,'codigo'), descricao: celN(r,'descricao'),
      qtde: Number(idxN['qtde'] != null ? r[idxN['qtde']]||0 : 0),
      endereco: celN(r,'endereco'), obs: celN(r,'obs'), usuario: celN(r,'usuario','user'),
    }));
  } catch(e) { return { ok: false, mensagem: 'Erro: ' + e.message }; }
}

// Lê o saldo já calculado na coluna ESTOQUE_ATUAL do Cadastro_PA (coluna U),
// em vez de somar ESTOQUE_INICIAL + histórico inteiro de Movimento_PA.
function obterSaldoPA(codigo) {
  try {
    const sh   = _sh(NOME_ABA_CADASTRO_PA);
    if (!sh) return 0;
    const data = sh.getDataRange().getValues();
    const h    = data[0];
    const iCOD = _idxCol(h, 'CÓDIGO') >= 0 ? _idxCol(h, 'CÓDIGO') : _idxCol(h, 'CODIGO');
    const iEA  = _idxCol(h, 'ESTOQUE_ATUAL');
    if (iCOD < 0 || iEA < 0) return 0;
    const row = data.slice(1).find(r => String(r[iCOD]||'').trim() === String(codigo).trim());
    return row ? Number(row[iEA]||0) : 0;
  } catch(e) { return 0; }
}

// Endereço é fixo por item (Cadastro_PA, coluna F) — lê só o Cadastro_PA,
// sem tocar em Movimento_PA. Mesmo formato de resposta de antes.
function obterEnderecosSaldoPA(codigo) {
  try {
    const sh = _sh(NOME_ABA_CADASTRO_PA);
    if (!sh) return { saldo: 0, enderecos: [] };
    const data = sh.getDataRange().getValues();
    const h    = data[0];
    const iCOD = _idxCol(h, 'CÓDIGO') >= 0 ? _idxCol(h, 'CÓDIGO') : _idxCol(h, 'CODIGO');
    const iEND = _idxCol(h, 'Endereço') >= 0 ? _idxCol(h, 'Endereço') : _idxCol(h, 'ENDEREÇO_PA');
    const iEA  = _idxCol(h, 'ESTOQUE_ATUAL');
    if (iCOD < 0) return { saldo: 0, enderecos: [] };

    const endereco = iEND >= 0 ? String(row[iEND]||'').trim() : '';
    return { saldo, enderecos: endereco ? [{ endereco, saldo }] : [] };
  } catch(e) { return { saldo: 0, enderecos: [] }; }

//  gravarMovimentoPA — com baixa automática de insumos via Estrutura
// ════════════════════════════════════════════════════════════════════════════
// ════════════════════════════════════════════════════════════════════════════
//  gravarMovimentoPA — otimizado: 2 leituras totais + idempotência por chave
// ════════════════════════════════════════════════════════════════════════════
function gravarMovimentoPA(dados) {
  try {
    const codigo   = String(dados.codigo||'').trim();
    const isSaida  = String(dados.tipo||'').toUpperCase() === 'SAIDA';
    const qtde     = Number(dados.qtde||0);
    const chaveIdem= String(dados.chaveIdem||'').trim(); // proteção contra dupla gravação

    if (!codigo) return { ok: false, mensagem: 'Código PA não informado.' };
    if (!qtde || qtde <= 0) return { ok: false, mensagem: 'Quantidade inválida.' };

    // ── Leitura 1: Cadastro_PA — situação + endereço numa leitura só ──────
    const infoCadPA = _mapaCadastroInfo(NOME_ABA_CADASTRO_PA);
    const itemCadPA = infoCadPA[codigo];
    if (itemCadPA === undefined) return { ok: false, mensagem: `Item PA '${codigo}' não encontrado.` };
    if (itemCadPA.situacao === 'INATIVO')  return { ok: false, mensagem: `Item PA '${codigo}' está INATIVO.` };

    // ── Leitura 2: Movimento_PA (saldo + idempotência) ────────────────────
    const shPA    = _sh(NOME_ABA_MOVIMENTO_PA);
    if (!shPA) return { ok: false, mensagem: 'Aba Movimento_PA não encontrada.' };
    const dadosPA = shPA.getDataRange().getValues();
    const headerPA= dadosPA[0];

    // Checa idempotência — se já existe um registro com essa chave, retorna ok sem gravar
    if (chaveIdem) {
      const iChave = headerPA.findIndex(v => _norm(v) === 'chave');
      if (iChave >= 0 && dadosPA.slice(1).some(r => String(r[iChave]||'').trim() === chaveIdem))
        return { ok: true, gravados: 0, jaGravado: true };
    }

    // Checa duplicidade de negócio — mesma OP real (não ADJ- manual) + código já baixados
    // Protege contra reabrir e confirmar a baixa da mesma OP duas vezes (chaveIdem muda a cada clique)
    const opStr = String(dados.op||'').trim();
    if (isSaida && opStr && !/^ADJ-/i.test(opStr)) {
      const iOPcol  = _idxCol(headerPA, 'op');
      const iCodCol = _idxCol(headerPA, 'codigo');
      const iTipoCol= _idxCol(headerPA, 'tipo');
      if (iOPcol >= 0 && iCodCol >= 0 && iTipoCol >= 0) {
        const jaBaixado = dadosPA.slice(1).some(r =>
          String(r[iOPcol]||'').trim() === opStr &&
          String(r[iCodCol]||'').trim() === codigo &&
          String(r[iTipoCol]||'').trim().toUpperCase() === 'SAIDA'
        );
        if (jaBaixado) return { ok: true, gravados: 0, jaGravado: true };
      }
    }

    // Checa duplicidade de negócio — mesmo código + tipo + qtde gravado nos últimos 90s.
    // Protege contra duplo-clique no botão Confirmar antes do modal fechar (a chaveIdem muda
    // a cada clique, e o OP ADJ- só tem granularidade de minuto — pode nem bater entre os cliques).
    {
      const iCodCol2  = _idxCol(headerPA, 'codigo');
      const iTipoCol2 = _idxCol(headerPA, 'tipo');
      const iQtdeCol2 = _idxCol(headerPA, 'qtde');
      const iDataCol2 = _idxCol(headerPA, 'data');
      if (iCodCol2 >= 0 && iTipoCol2 >= 0 && iQtdeCol2 >= 0 && iDataCol2 >= 0) {
        const agoraMs  = Date.now();
        const tipoAlvo = isSaida ? 'SAIDA' : 'ENTRADA';
        const jaGravado = dadosPA.slice(1).some(r => {
          if (String(r[iCodCol2]||'').trim() !== codigo) return false;
          if (String(r[iTipoCol2]||'').trim().toUpperCase() !== tipoAlvo) return false;
          if (Number(r[iQtdeCol2]||0) !== qtde) return false;
          const t = new Date(r[iDataCol2]).getTime();
          return !isNaN(t) && Math.abs(agoraMs - t) <= 90000;
        });
        if (jaGravado) return { ok: true, gravados: 0, jaGravado: true };
      }
    }

    const saldosPA   = _saldosEmMemoria(dadosPA, headerPA);
    const linhasAntes= dadosPA.length;
    const agora      = new Date();
    const endereco   = itemCadPA.endereco || '';
    const base       = {
      data: agora, qrcode: dados.qrcode||'', tipo: dados.tipo||'',
      op: dados.op||'', codigo, descricao: dados.descricao||'',
      obs: dados.obs||'', usuario: dados.usuario||'',
      computer: dados.computer||'', localizacao: dados.localizacao||'',
      ip: dados.ip||'', chave: chaveIdem,
    };

    if (isSaida) {
      const saldoTotal = saldosPA[codigo] || 0;
      if (qtde > saldoTotal) return { ok: false, mensagem: `Saldo PA insuficiente para '${codigo}'. Disponível: ${saldoTotal}.` };
    }
    const id  = Utilities.getUuid().substring(0, 8);
    const ids = [id];
    shPA.appendRow(_montarLinhaMovimento(headerPA, Object.assign({}, base, { id, qtde, endereco })));

    SpreadsheetApp.flush();

    // ── Leitura 3 (só insumos, só se saída): Estrutura + Movimento ────────
    // prepararBaixaInsumos é chamado SEPARADAMENTE pelo frontend
    // gravarMovimentoPA não baixa insumos automaticamente — o frontend controla
    return { ok: true, gravados: ids.length };
  } catch(e) { return { ok: false, mensagem: 'Erro ao gravar PA: ' + e.message }; }
}

// ════════════════════════════════════════════════════════════════════════════
//  pagarOPTriangularPA — com baixa automática de insumos via Estrutura
// ════════════════════════════════════════════════════════════════════════════
function pagarOPTriangularPA(dados) {
  try {
    const codOrigem  = String(dados.codOrigem  || '').trim();
    const codDestino = String(dados.codDestino || '').trim();
    const qtde       = Number(dados.qtde || 0);
    const op         = String(dados.op   || '').trim();
    const usuario    = dados.usuario    || '';
    const computer   = dados.computer   || '';
    const localizacao= dados.localizacao|| '';
    const ip         = dados.ip         || '';
    const chaveIdem  = String(dados.chaveIdem || '').trim();

    if (!codOrigem)  return { ok: false, mensagem: 'Código de origem não informado.' };
    if (!codDestino) return { ok: false, mensagem: 'Código de destino não informado.' };
    if (codOrigem === codDestino) return { ok: false, mensagem: 'Origem e destino não podem ser iguais.' };
    if (!qtde || qtde <= 0) return { ok: false, mensagem: 'Quantidade inválida.' };

    const mapa = _mapaStatusCadastroPA();
    if (mapa[codOrigem]  === undefined) return { ok: false, mensagem: `PA origem '${codOrigem}' não cadastrado.` };
    if (mapa[codOrigem]  === 'INATIVO')  return { ok: false, mensagem: `PA origem '${codOrigem}' INATIVO.` };
    if (mapa[codDestino] === undefined) return { ok: false, mensagem: `PA destino '${codDestino}' não cadastrado.` };
    if (mapa[codDestino] === 'INATIVO')  return { ok: false, mensagem: `PA destino '${codDestino}' INATIVO.` };

    const sh = _sh(NOME_ABA_MOVIMENTO_PA);
    if (!sh) return { ok: false, mensagem: 'Aba Movimento_PA não encontrada.' };

    // ── Leitura ÚNICA de Movimento_PA: saldo + idempotência + header ─────
    const dadosSh     = sh.getDataRange().getValues();
    const header      = dadosSh[0];
    const linhasAntes = dadosSh.length;
    const saldoMapa   = _saldosEmMemoria(dadosSh, header);
    // Inclui estoque inicial para manter paridade com obterSaldoPA()
    const estoqueIni  = _estoqueInicialPA(codOrigem);
    const saldoOrigem = estoqueIni + (saldoMapa[codOrigem]||0);
    if (qtde > saldoOrigem) return { ok: false, mensagem: `Saldo insuficiente para ${codOrigem}. Disponível: ${saldoOrigem}.` };

    if (chaveIdem) {
      const iChkIdx = header.findIndex(v => _norm(v) === 'chave');
      if (iChkIdx >= 0 && dadosSh.slice(1).some(r => String(r[iChkIdx]||'').trim() === chaveIdem))
        return { ok: true, gravados: 0, jaGravado: true };
    }

    // Checa duplicidade de negócio — mesma OP real (não ADJ- manual) já baixada para o código origem
    if (op && !/^ADJ-/i.test(op)) {
      const iOPcol  = _idxCol(header, 'op');
      const iCodCol = _idxCol(header, 'codigo');
      const iTipoCol= _idxCol(header, 'tipo');
      if (iOPcol >= 0 && iCodCol >= 0 && iTipoCol >= 0) {
        const jaBaixado = dadosSh.slice(1).some(r =>
          String(r[iOPcol]||'').trim()  === op &&
          String(r[iCodCol]||'').trim() === codOrigem &&
          String(r[iTipoCol]||'').trim().toUpperCase() === 'SAIDA'
        );
        if (jaBaixado) return { ok: true, gravados: 0, jaGravado: true };
      }
    }

    // Busca descrições e endereços
    const shCad   = _sh(NOME_ABA_CADASTRO_PA);
    const dataCad = shCad ? shCad.getDataRange().getValues() : [];
    const hCad    = dataCad[0] || [];
    const iCodCad = _idxCol(hCad, 'CÓDIGO');
    const iDescCad= hCad.findIndex(v => _norm(v) === 'descricao');
    const iEndCad = hCad.findIndex(v => _norm(v) === 'endereco');
    const findRow = (cod) => dataCad.slice(1).find(r => String(r[iCodCad]||'').trim() === cod);
    const rowOri  = findRow(codOrigem);
    const rowDes  = findRow(codDestino);
    const descOri = (rowOri && iDescCad >= 0) ? String(rowOri[iDescCad]||'') : '';
    const descDes = (rowDes && iDescCad >= 0) ? String(rowDes[iDescCad]||'') : '';
    const endOri  = (rowOri && iEndCad  >= 0) ? String(rowOri[iEndCad] ||'') : '';
    const endDes  = (rowDes && iEndCad  >= 0) ? String(rowDes[iEndCad] ||'') : '';

    const agora  = new Date();
    const base   = { qtde, op, usuario, computer, localizacao, ip, qrcode: '', chave: chaveIdem };

    // Grava os 3 movimentos triangulares em uma operação setValues
    const linhas = [
      _montarLinhaMovimento(header, Object.assign({}, base, { id: Utilities.getUuid().substring(0,8), data: agora, tipo: 'SAIDA',   codigo: codOrigem,  descricao: descOri, endereco: endOri, obs: `Triangular SAIDA VIA OP: ${op} PARA: ${codDestino}` })),
      _montarLinhaMovimento(header, Object.assign({}, base, { id: Utilities.getUuid().substring(0,8), data: agora, tipo: 'ENTRADA', codigo: codDestino, descricao: descDes, endereco: endDes, obs: `Triangular ENTRADA VIA OP: ${op} DE: ${codOrigem}` })),
      _montarLinhaMovimento(header, Object.assign({}, base, { id: Utilities.getUuid().substring(0,8), data: agora, tipo: 'SAIDA',   codigo: codDestino, descricao: descDes, endereco: endDes, obs: `Triangular SAIDA VIA OP: ${op}` })),
    ];
    sh.getRange(linhasAntes + 1, 1, linhas.length, header.length).setValues(linhas);
    SpreadsheetApp.flush();

    // Baixa dos insumos do produto DESTINO NÃO é feita aqui — o frontend controla,
    // via modal de confirmação de insumos (mesmo padrão de gravarMovimentoPA).
    // Antes havia baixa automática aqui + baixa manual pelo modal, duplicando os insumos.
    return { ok: true, gravados: 3 };
  } catch(e) { return { ok: false, mensagem: 'Erro Triangular: ' + e.message }; }
}

// ════════════════════════════════════════════════════════════════════════════
//  OPs — retorna pedido, origem, estado (ABERTA/PAGO/EM_BAIXA), foto, zpl
// ════════════════════════════════════════════════════════════════════════════
function listarOPS() {
  try {
    const sh = _sh(NOME_ABA_OPS);
    if (!sh) return { ok: false, mensagem: 'Aba OPS não encontrada.' };
    const data = sh.getDataRange().getValues();
    if (data.length < 2) return [];
    const h   = data[0];
    const idx = {};
    h.forEach((v,i) => { idx[String(v||'').trim().toUpperCase()] = i; });
    const get = (row, col) => col != null && col >= 0 ? String(row[col]||'').trim() : '';

    // Índices das colunas — compatível com nomes maiúsculos/minúsculos
    const iFoto    = idx['FOTO'] ?? -1; // foto já mora na própria OPS — sem
                                        // precisar ler Cadastro/Cadastro_PA
    const iOP      = idx['OP']      ?? idx['N_OP']     ?? -1;
    const iPedido  = idx['PEDIDO']  ?? idx['N_PEDIDO'] ?? -1;
    const iCodigo  = idx['CODIGO']  ?? idx['CÓDIGO']   ?? -1;
    const iDesc    = idx['DESCRICAO']?? idx['DESCRIÇÃO']?? -1;
    const iQtde    = idx['QTDE']    ?? idx['QUANTIDADE']?? -1;
    const iOrigem  = idx['ORIGEM']  ?? -1;
    const iPosicao = idx['POSICAO'] ?? idx['POSIÇÃO']  ?? -1;
    const iObs     = idx['OBS']     ?? idx['OBSERVACAO']?? -1;
    const iZPL     = idx['ZPL']        ?? -1;
    const iMetal   = idx['METALIZADA'] ?? -1;
    const iSilk    = idx['SILK']       ?? -1;
    const iPago    = idx['PAGO']       ?? -1; // coluna com fórmula (ARRAYFORMULA/COUNTIF)
                                               // que já verifica na planilha se a OP tem
                                               // movimento — substitui escanear Movimento/
                                               // Movimento_PA inteiros aqui no código

    return data.slice(1)
      .filter(r => {
        const pos = get(r, iPosicao).toUpperCase();
        return pos === 'ESTOQUE' || pos === 'PAGO' || pos === 'EM BAIXA';
      })
      .map(r => {
        const opNum  = get(r, iOP);
        const codigo = get(r, iCodigo);
        const origem = get(r, iOrigem).toUpperCase(); // 'INSUMO' | 'PA'
        const isPA   = origem === 'PA';
        const posicaoPlanilha = get(r, iPosicao).toUpperCase();

        // Estado: PAGO se a coluna PAGO (fórmula) diz, EM_BAIXA se planilha diz, senão ABERTA
        let estado;
        if (get(r, iPago).toUpperCase() === 'PAGO') estado = 'PAGO';
        else if (posicaoPlanilha === 'EM BAIXA')    estado = 'EM_BAIXA';
        else                                         estado = 'ABERTA';

        return {
          op:       opNum,
          pedido:   get(r, iPedido),
          codigo,
          descricao: get(r, iDesc),
          qtde:     Number(r[iQtde < 0 ? 0 : iQtde]||0),
          origem,                          // 'INSUMO' | 'PA'
          isPA,
          estado,                          // 'ABERTA' | 'PAGO' | 'EM_BAIXA'
          obs:      get(r, iObs),
          zpl:      get(r, iZPL),
          zplMetal: get(r, iMetal),
          silk:     get(r, iSilk),
          fotoUrl:  get(r, iFoto),
        };
      });
  } catch(e) { return { ok: false, mensagem: 'Erro: ' + e.message }; }
}

// ════════════════════════════════════════════════════════════════════════════
//  BOM
// ════════════════════════════════════════════════════════════════════════════
function prepararEListarBOM(opNumero, dadosOPC) {
  // dadosOPC = { codigoPA, qtde } — quando passados pelo frontend, dispensa ler a aba OPS
  try {
    let codigoPA, qtdePedido;

    if (dadosOPC?.codigoPA && Number(dadosOPC?.qtde) > 0) {
      // ── Fast path: produto e quantidade já vieram do frontend ────────────
      codigoPA   = String(dadosOPC.codigoPA).trim();
      qtdePedido = Number(dadosOPC.qtde);
    } else {
      // ── Fallback: busca na aba OPS ───────────────────────────────────────
      const shOPS = _sh(NOME_ABA_OPS);
      if (!shOPS) return { ok: false, mensagem: 'Aba OPS não encontrada.' };
      const dataOPS = shOPS.getDataRange().getValues();
      const hOPS    = dataOPS[0];
      const idxOPS  = {};
      hOPS.forEach((v,i) => { idxOPS[String(v||'').trim().toUpperCase()] = i; });
      const iOP     = idxOPS['OP']     ?? idxOPS['N_OP']      ?? -1;
      const iCodigo = idxOPS['CODIGO'] ?? idxOPS['CÓDIGO']    ?? -1;
      const iQtde   = idxOPS['QTDE']   ?? idxOPS['QUANTIDADE'] ?? -1;
      const opStr   = String(opNumero).trim();
      const rowOP   = dataOPS.slice(1).find(r => String(r[iOP]||'').trim() === opStr
                                               || String(r[iOP]||'').trim() === String(Number(opStr)));
      if (!rowOP) return { ok: false, mensagem: `OP ${opStr} não encontrada na planilha OPS.` };
      codigoPA   = iCodigo >= 0 ? String(rowOP[iCodigo]||'').trim() : '';
      qtdePedido = iQtde   >= 0 ? Number(rowOP[iQtde]  ||0) : 0;
      if (!codigoPA) return { ok: false, mensagem: 'Código do produto não encontrado na OP.' };
    }

    // ── Leitura 2: Estrutura de Produtos ─────────────────────────────────
    const shEst  = _sh(NOME_ABA_ESTRUTURA);
    if (!shEst) return { ok: false, mensagem: 'Aba "Estrutura de Produtos" não encontrada.' };
    const dataEst  = shEst.getDataRange().getValues();
    const hEst     = dataEst[0];
    const iCodPA   = _idxCol(hEst, 'CODIGO_PRODUTO');
    const iCodIns  = _idxCol(hEst, 'CODIGO_INSUMO');
    const iDescIns = _idxCol(hEst, 'DESCRICAO_INSUMO');
    const iQtdeEst = _idxCol(hEst, 'QTDE_UTILIZADA');
    const iOrdem   = _idxCol(hEst, 'ORDEM');
    const iTipoMov = _idxCol(hEst, 'TIPO_MOV'); // coluna opcional na estrutura
    if (iCodPA < 0 || iCodIns < 0 || iQtdeEst < 0)
      return { ok: false, mensagem: 'Colunas obrigatórias não encontradas na Estrutura de Produtos.' };

    // ── Leitura 3: mapa PA para inferir tipoMov quando não há coluna ─────
    const mapaPA = iTipoMov < 0 ? _mapaStatusCadastroPA() : null;

    const cod   = codigoPA.toUpperCase();
    const itens = dataEst.slice(1)
      .filter(r => String(r[iCodPA]||'').trim().toUpperCase() === cod && String(r[iCodIns]||'').trim())
      .map(r => {
        const qtdBom    = Number(r[iQtdeEst]||0);
        const qtdBaixar = qtdBom * qtdePedido;
        const codIns    = String(r[iCodIns]||'').trim();
        let tipoMov;
        if (iTipoMov >= 0) {
          tipoMov = String(r[iTipoMov]||'').trim().toUpperCase() || 'INSUMO';
        } else {
          tipoMov = (mapaPA[codIns] !== undefined) ? 'PA' : 'INSUMO';
        }
        return {
          ordem:        iOrdem >= 0 ? String(r[iOrdem]||'').trim() : '',
          codigo:       codIns,
          descricao:    iDescIns >= 0 ? String(r[iDescIns]||'').trim() : '',
          qtdBom,
          qtdPedido: qtdePedido,
          qtdBaixar,
          saldoEstoque: 0,
          permitido:    true,
          tipoMov,
        };
      });

    // Fallback: produto não está em Estrutura de Produtos — tenta aba BOM (fórmulas + sleep)
    if (itens.length === 0) {
      const shBom = _sh(NOME_ABA_BOM);
      if (shBom) {
        shBom.getRange('A2').setValue(String(opNumero).trim());
        SpreadsheetApp.flush();

        // As colunas QTDE_BOM/QTDE_PEDIDO dependem de fórmulas que recalculam
        // após o flush. Em vez de um sleep fixo de 3s (sempre pago, mesmo
        // quando a planilha já recalculou em 300ms), faz polling curto e sai
        // assim que duas leituras seguidas derem o mesmo resultado — mesmo
        // teto de espera (10×300ms=3s) do sleep antigo, mas normalmente sai
        // bem antes.
        let dataBom = shBom.getDataRange().getValues();
        for (let tentativa = 0; tentativa < 10; tentativa++) {
          Utilities.sleep(300);
          const proxima = shBom.getDataRange().getValues();
          if (JSON.stringify(proxima) === JSON.stringify(dataBom)) { dataBom = proxima; break; }
          dataBom = proxima;
        }
        if (dataBom.length >= 2) {
          const hBom = dataBom[0];
          const idxB = {}, idxBN = {};
          hBom.forEach((v,i) => { idxB[String(v).trim().toUpperCase()]=i; const k=_norm(v); if(!idxBN[k]) idxBN[k]=i; });
          const iO  = idxB['ORDEM']??0,  iC = idxB['CODIGO']??1,  iD  = idxB['DESCRICAO']??2;
          const iOP = idxB['OP']??3,     iB = idxB['QTDE_BOM']??idxB['QTD_BOM']??5;
          const iP  = idxB['QTDE_PEDIDO']??idxB['QTD_PEDIDO']??6;
          const iTM = idxBN['tipomov']??idxBN['tipo mov']??idxB['TIPO_MOV']??-1;
          const opN = String(opNumero).trim(), opNorm = opN.replace(/[^0-9a-zA-Z]/g,'').toLowerCase();
          dataBom.slice(1)
            .filter(r => {
              const opPlan = String(r[iOP]||'').trim();
              return (opPlan === opN || opPlan.replace(/[^0-9a-zA-Z]/g,'').toLowerCase() === opNorm)
                     && String(r[iC]||'').trim() !== '';
            })
            .forEach(r => {
              const qtdBom = Number(r[iB]||0), qtdPedido = Number(r[iP]||0);
              const tipoMov = iTM >= 0 ? String(r[iTM]||'').trim().toUpperCase() || 'INSUMO' : 'INSUMO';
              itens.push({ ordem: String(r[iO]||'').trim(), codigo: String(r[iC]||'').trim(),
                descricao: String(r[iD]||'').trim(), qtdBom, qtdPedido, qtdBaixar: qtdBom * qtdPedido,
                saldoEstoque: 0, permitido: true, tipoMov });
            });
        }
      }
    }

    return { ok: true, itens };
  } catch(e) { return { ok: false, mensagem: 'Erro BOM: ' + e.message }; }
}

// ════════════════════════════════════════════════════════════════════════════
//  prepararBOMComSaldo — BOM + saldo atual de cada item em memória
//  Igual ao prepararEListarBOM mas enriquece cada item com saldoAtual
//  para exibir no modal com checkbox + quantidade ajustável
// ════════════════════════════════════════════════════════════════════════════
function prepararBOMComSaldo(dados) {
  try {
    const opNumero = typeof dados === 'object' ? (dados.op || dados) : dados;
    const dadosOPC = typeof dados === 'object' ? { codigoPA: dados.codigoPA, qtde: dados.qtde } : null;
    // Passo 1: carrega estrutura de produto (sem sleep — lê OPS ou usa dados do frontend)
    const resBom = prepararEListarBOM(opNumero, dadosOPC);
    if (!resBom.ok) return resBom;
    if (!resBom.itens.length) return { ok: true, itens: [] };

    // Passo 2: lê saldos (ESTOQUE_ATUAL) de todos os Insumos e PA de uma vez —
    // mais rápido que escanear Movimento/Movimento_PA, e mesma fonte usada
    // em obterSaldo/obterSaldoPA.
    const saldosIns = _mapaEstoqueAtualCadastro(NOME_ABA_CADASTRO);
    const saldosPA  = _mapaEstoqueAtualCadastro(NOME_ABA_CADASTRO_PA);

    // Passo 3: enriquece cada item com saldoAtual
    const itens = resBom.itens.map(it => {
      const saldoAtual = it.tipoMov === 'PA'
        ? (saldosPA[it.codigo] || 0)
        : (saldosIns[it.codigo] || 0);
      return {
        ...it,
        saldoAtual,
        suficiente: saldoAtual >= it.qtdBaixar,
      };
    });

    return { ok: true, itens };
  } catch(e) { return { ok: false, mensagem: 'Erro BOM com saldo: ' + e.message }; }
}

function baixarBOMComTipo(movimentos, dadosBase) {
  try {
    if (!movimentos?.length) return { ok: false, mensagem: 'Nenhum item para baixar.' };
    const op=String(dadosBase.op||'').trim(), usuario=String(dadosBase.usuario||''), computer=String(dadosBase.computer||''), localizacao=String(dadosBase.localizacao||''), ip=String(dadosBase.ip||''), obs=String(dadosBase.obs||'');
    const chaveIdem=String(dadosBase.chaveIdem||'').trim();
    const listaInsumo=[], listaPA=[];
    for (const item of movimentos) {
      const tipo = String(item.tipoMov||'INSUMO').trim().toUpperCase();
      const mov  = { codigo:String(item.codigo||'').trim(), descricao:String(item.descricao||''), qtde:Number(item.qtdBaixar||0), tipo:'SAIDA', op, usuario, computer, localizacao, ip, obs:obs||'Baixa BOM — OP '+op, qrcode:'', endereco:'', chave:chaveIdem };
      if (tipo==='PA') listaPA.push(mov); else listaInsumo.push(mov);
    }
    const resultados = { ok:true, insumos:null, pa:null, erros:[] };
    if (listaInsumo.length>0) { const r=gravarMovimentosEmLote(listaInsumo); resultados.insumos=r; if(!r.ok) resultados.erros.push('INSUMOS: '+r.mensagem); }
    if (listaPA.length>0)     { const r=gravarMovimentosEmLotePA(listaPA);   resultados.pa=r;      if(!r.ok) resultados.erros.push('PA: '+r.mensagem); }
    if (resultados.erros.length>0) { resultados.ok=false; resultados.mensagem=resultados.erros.join('\n'); }
    else { const g=(resultados.insumos?.gravados||0)+(resultados.pa?.gravados||0); resultados.mensagem=`Baixa concluída: ${g} movimento(s).`; resultados.gravados=g; }
    return resultados;
  } catch(e) { return { ok: false, mensagem: 'Erro BOM: ' + e.message }; }
}

// Endereço agora é 1 campo fixo do Cadastro (não mais um saldo por endereço
// em Movimento) — "mudar endereço" virou uma edição simples desse campo,
// sem gerar movimento nenhum.
function mudarEndereco(dados) {
  try {
    const codigo = String(dados.codigo || '').trim();
    const novoEndereco = String(dados.novoEndereco ?? dados.endDestino ?? '').trim();

    const h    = data[0];
    const iCOD = _idxCol(h, 'CÓDIGO');
    const iEND = _idxCol(h, 'Endereço');

    if (rowIdx < 0) return { ok: false, mensagem: `Item '${codigo}' não encontrado.` };
    sh.getRange(rowIdx + 2, iEND + 1).setValue(novoEndereco);
    return { ok: true, mensagem: 'Endereço atualizado!' };
  } catch(e) { return { ok: false, mensagem: 'Erro: ' + e.message }; }
}

function listarEmbarque() {
  try {
    const shEmb=_sh(NOME_ABA_EMBARQUE);
    if (!shEmb) return { ok:false, mensagem:'Aba Embarque não encontrada.' };
    const lastRow=shEmb.getLastRow();
    if (lastRow<2) return { ok:true, itens:[] };
    const dados=shEmb.getRange(2,17,lastRow-1,26).getValues();
    const shMov=_sh(NOME_ABA_MOVIMENTO);
    const idsLancados=new Set();
    if (shMov&&shMov.getLastRow()>1) {
      shMov.getRange(2,1,shMov.getLastRow()-1,1).getValues().forEach(r=>{ if(r[0]) idsLancados.add(String(r[0]).trim()); });
    }
    const itens=dados.filter(r=>String(r[2]||'').trim()!=='').map(r=>({
      codigo:String(r[2]||'').trim(), descricao:String(r[3]||'').trim(), qtde:Number(r[4]||0),
      peso:Number(r[5]||0), usdUnit:Number(r[6]||0), mov_id:String(r[13]||'').trim(),
      mov_codigo:String(r[18]||'').trim(), mov_qtde:Number(r[19]||0), mov_tipo:String(r[16]||'').trim(),
      mov_op:String(r[17]||'').trim(), mov_obs:String(r[22]||'').trim(), mov_usuario:String(r[23]||'').trim(),
      jaLancado:idsLancados.has(String(r[13]||'').trim()),
    }));
    return { ok:true, itens };
  } catch(e) { return { ok:false, mensagem:'Erro embarque: '+e.message }; }
}

function gravarEntradaDI(item) {
  try {
    if (!item?.mov_id||!item?.mov_codigo) return { ok:false, mensagem:'Dados insuficientes.' };
    const shMov=_sh(NOME_ABA_MOVIMENTO);
    if (!shMov) return { ok:false, mensagem:'Aba Movimento não encontrada.' };
    const lastRowMov=shMov.getLastRow();
    if (lastRowMov>1) {
      const ids=shMov.getRange(2,1,lastRowMov-1,1).getValues();
      if (ids.some(r=>String(r[0]).trim()===item.mov_id)) return { ok:false, mensagem:`ID '${item.mov_id}' já lançado.` };
    }
    const headerMov=shMov.getRange(1,1,1,shMov.getLastColumn()).getValues()[0];
    const linha=_montarLinhaMovimento(headerMov,{ id:item.mov_id, data:new Date(), qrcode:item.mov_qrcode||'', tipo:item.mov_tipo||'ENTRADA', op:item.mov_op||'', codigo:item.mov_codigo, descricao:item.descricao||'', qtde:Number(item.mov_qtde||0), endereco:item.mov_endereco||'', chave:item.mov_chave||'', obs:item.mov_obs||'', usuario:item.mov_usuario||'', localizacao:item.mov_loc||'', computer:'', ip:item.mov_ip||'' });
    shMov.getRange(lastRowMov+1,1,1,headerMov.length).setValues([linha]);
    SpreadsheetApp.flush();
    return { ok:true };
  } catch(e) { return { ok:false, mensagem:'Erro DI: '+e.message }; }
}

function _printNode(printerId, titulo, conteudo) {
  try {
    const res=UrlFetchApp.fetch('https://api.printnode.com/printjobs',{ method:'post', contentType:'application/json', payload:JSON.stringify({ printerId, title:titulo, contentType:'raw_base64', content:Utilities.base64Encode(conteudo.toString(),Utilities.Charset.UTF_8), source:'GAS-Estoque' }), muteHttpExceptions:true, headers:{ 'Authorization':'Basic '+Utilities.base64Encode(PRINTNODE_API_KEY+':') } });
    const code=res.getResponseCode();
    return code>=200&&code<300 ? { ok:true } : { ok:false, mensagem:'PrintNode HTTP '+code };
  } catch(e) { return { ok:false, mensagem:e.message }; }
}

// ════════════════════════════════════════════════════════════════════════════
//  prepararBaixaInsumos — leitura única: Estrutura + Movimento fundidos
// ════════════════════════════════════════════════════════════════════════════
function prepararBaixaInsumos(codigoPA, qtdePABaixado) {
  try {
    // ── Leitura 1: Estrutura de Produtos ─────────────────────────────────
    const shEst = _sh(NOME_ABA_ESTRUTURA);
    if (!shEst) return { ok: false, mensagem: 'Aba "Estrutura de Produtos" não encontrada.' };
    const dataEst  = shEst.getDataRange().getValues();
    const headerEst= dataEst[0];
    const iCodPA   = _idxCol(headerEst, 'CODIGO_PRODUTO');
    const iCodIns  = _idxCol(headerEst, 'CODIGO_INSUMO');
    const iDescIns = _idxCol(headerEst, 'DESCRICAO_INSUMO');
    const iQtdeEst = _idxCol(headerEst, 'QTDE_UTILIZADA');
    const iOrdem   = _idxCol(headerEst, 'ORDEM');
    if (iCodPA < 0 || iCodIns < 0 || iQtdeEst < 0)
      return { ok: false, mensagem: 'Colunas obrigatórias não encontradas na Estrutura de Produtos.' };

    const cod  = String(codigoPA||'').trim().toUpperCase();
    const qtde = Number(qtdePABaixado) || 1;
    const itensEst = dataEst.slice(1)
      .filter(r => String(r[iCodPA]||'').trim().toUpperCase() === cod && String(r[iCodIns]||'').trim())
      .map(r => ({
        codigoInsumo:  String(r[iCodIns]  ||'').trim(),
        descricao:     iDescIns >= 0 ? String(r[iDescIns]||'').trim() : '',
        qtdeUtilizada: Number(r[iQtdeEst] || 0),
        ordem:         iOrdem  >= 0 ? String(r[iOrdem]   ||'').trim() : '',
      }));

    if (!itensEst.length)
      return { ok: true, itens: [], aviso: `Nenhum insumo cadastrado na Estrutura de Produtos para ${cod}.` };

    // ── Leitura 2: ESTOQUE_ATUAL de todos os insumos do Cadastro ──────────
    const saldosMap = _mapaEstoqueAtualCadastro(NOME_ABA_CADASTRO);

    const itens = itensEst.map(item => {
      const qtdeNecessaria = item.qtdeUtilizada * qtde;
      const saldoAtual     = saldosMap[item.codigoInsumo] || 0;
      return { ...item, qtdeNecessaria, saldoAtual, suficiente: saldoAtual >= qtdeNecessaria };
    });

    return { ok: true, itens };
  } catch(e) { return { ok: false, mensagem: 'Erro ao preparar insumos: ' + e.message }; }
}

// ════════════════════════════════════════════════════════════════════════════
//  gravarBaixaInsumos — setValues em bloco (não appendRow por linha)
//  Recebe: itens = [{ codigoInsumo, descricao, qtdeBaixar }]
//          dadosBase = { op, usuario, chaveIdem, ... }
// ════════════════════════════════════════════════════════════════════════════
function gravarBaixaInsumos(itens, dadosBase) {
  try {
    if (!itens?.length) return { ok: true, gravados: 0, avisos: ['Nenhum insumo para baixar.'] };

    const chaveIdem = String(dadosBase.chaveIdem||'').trim();
    const paraGravar = itens.filter(it => Number(it.qtdeBaixar||0) > 0);
    if (!paraGravar.length) return { ok: true, gravados: 0, avisos: ['Nenhum item com quantidade > 0.'] };

    const shMov    = _sh(NOME_ABA_MOVIMENTO);
    if (!shMov) return { ok: false, mensagem: 'Aba Movimento não encontrada.' };
    const dadosSh  = shMov.getDataRange().getValues();
    const header   = dadosSh[0];

    // Idempotência: se já existe chave gravada, retorna ok sem duplicar
    if (chaveIdem) {
      const iChave = header.findIndex(v => _norm(v) === 'chave');
      if (iChave >= 0 && dadosSh.slice(1).some(r => String(r[iChave]||'').trim() === chaveIdem))
        return { ok: true, gravados: 0, jaGravado: true, avisos: [] };
    }

    // 1 leitura só do Cadastro: situação + endereço + saldo (antes eram
    // 3 leituras separadas — _mapaEstoqueAtualCadastro + _mapaEnderecoCadastro
    // + _mapaStatusCadastro, cada uma relendo a aba inteira)
    const infoCad   = _mapaCadastroInfo(NOME_ABA_CADASTRO);
    const agora     = new Date();
    const avisos    = [];
    const linhas    = []; // acumula todas as linhas para gravar em um único setValues

    for (const it of paraGravar) {
      const cod = String(it.codigoInsumo || it.codigo || '').trim();
      if (!cod) continue;
      const item = infoCad[cod];
      if (item === undefined) { avisos.push(`'${cod}' não cadastrado — pulado.`); continue; }
      if (item.situacao === 'INATIVO')  { avisos.push(`'${cod}' INATIVO — pulado.`); continue; }
      const saldo = item.saldo || 0;
      if (saldo <= 0) { avisos.push(`Sem saldo para '${cod}' — pulado.`); continue; }
      const qtde = Math.min(Number(it.qtdeBaixar), saldo);
      if (qtde < Number(it.qtdeBaixar)) avisos.push(`Saldo insuficiente '${cod}': necessário ${it.qtdeBaixar}, disponível ${saldo}.`);

      linhas.push(_montarLinhaMovimento(header, {
        id: Utilities.getUuid().substring(0,8), data: agora,
        tipo: 'SAIDA', codigo: cod, descricao: it.descricao||'', qtde,
        endereco: item.endereco||'', op: dadosBase.op||'',
        obs: `Baixa insumos via PA — OP: ${dadosBase.op||''}`,
        usuario: dadosBase.usuario||'', computer: dadosBase.computer||'',
        localizacao: dadosBase.localizacao||'', ip: dadosBase.ip||'',
        qrcode:'', chave: chaveIdem,
      }));
    }

    if (!linhas.length) return { ok: true, gravados: 0, avisos };

    // Grava tudo em um único setValues — muito mais rápido que appendRow em loop
    const ultimaLinha = shMov.getLastRow();
    shMov.getRange(ultimaLinha + 1, 1, linhas.length, header.length).setValues(linhas);
    SpreadsheetApp.flush();

    return { ok: true, gravados: linhas.length, avisos };
  } catch(e) { return { ok: false, mensagem: 'Erro ao gravar insumos: ' + e.message }; }
}

// ════════════════════════════════════════════════════════════════════════════
//  IMPRESSÃO DE CARTÕES — G (grande) e P (grade A4)
//  Reutiliza os templates Cartao.html e CartaoGrade.html existentes
//  lista = [{ codigo, layout }]  onde layout = 'G' ou 'P'
// ════════════════════════════════════════════════════════════════════════════
function imprimirFilaCartaoInsumo(lista) {
  try {
    if (!lista?.length) return { ok: false, mensagem: 'Lista vazia.' };

    // Monta mapa de produtos
    const shCad   = _sh(NOME_ABA_CADASTRO);
    if (!shCad) return { ok: false, mensagem: 'Aba Cadastro não encontrada.' };
    const data    = shCad.getDataRange().getValues();
    const header  = data[0];
    const iCodigo   = _idxCol(header, 'CÓDIGO');
    const iFoto     = _idxCol(header, 'Endereço_Foto');
    const iDescricao= _idxCol(header, 'DESCRIÇÃO');
    const iObs      = _idxCol(header, 'obs');
    const iEndereco = _idxCol(header, 'Endereço');
    const mapaProd  = {};
    data.slice(1).forEach(r => {
      const cod = String(r[iCodigo] || '').trim();
      if (!cod) return;
      mapaProd[cod] = {
        img:      iFoto      >= 0 ? String(r[iFoto]      || '') : '',
        detalhe:  iDescricao >= 0 ? String(r[iDescricao] || '') : cod,
        obs:      iObs       >= 0 ? String(r[iObs]       || '') : '',
        endereco: iEndereco  >= 0 ? String(r[iEndereco]  || '') : '',
      };
    });

    const logoBase64 = _buscarImagemDriveBase64(ID_LOGO_FUTURA);
    const listaG = [], listaP = [];

    for (const item of lista) {
      const prod = mapaProd[item.codigo];
      if (!prod) { Logger.log('Insumo não encontrado: ' + item.codigo); continue; }
      const dadosFinal = {
        img:      prod.img ? _buscarImagemDriveBase64(prod.img) : '',
        logo:     logoBase64,
        detalhe:  prod.detalhe || item.codigo,
        enderecos:prod.endereco || '—',
        obs:      prod.obs || '',
        codigo:   item.codigo,
      };
      if (item.layout === 'G') listaG.push(dadosFinal);
      else                     listaP.push(dadosFinal);
    }

    if (!listaG.length && !listaP.length)
      return { ok: false, mensagem: 'Nenhum insumo localizado na planilha.' };

    const erros = [];
    const pdfs  = []; // PDF gerado é sempre devolvido pro frontend baixar/abrir
                       // localmente, independente do resultado do PrintNode
                       // (útil enquanto a impressora física está com defeito)
    if (listaG.length > 0) {
      try {
        const tmpl = HtmlService.createTemplateFromFile('Cartao');
        tmpl.lista = listaG;
        const b64  = Utilities.base64Encode(tmpl.evaluate().getBlob().getAs(MimeType.PDF).getBytes());
        pdfs.push({ nome: 'Insumos_G_' + listaG.length + '.pdf', base64: b64 });
        const res  = _enviarPDFParaPrintNode(b64, 'Insumos G (' + listaG.length + ')');
        if (!res.ok) erros.push('Layout G: ' + res.mensagem);
      } catch(eG) { erros.push('Layout G: ' + eG.message); }
    }
    if (listaP.length > 0) {
      try {
        const tmpl = HtmlService.createTemplateFromFile('CartaoGrade');
        tmpl.lista = listaP;
        const b64  = Utilities.base64Encode(tmpl.evaluate().getBlob().getAs(MimeType.PDF).getBytes());
        pdfs.push({ nome: 'Insumos_P_' + listaP.length + '.pdf', base64: b64 });
        const res  = _enviarPDFParaPrintNode(b64, 'Insumos P (' + listaP.length + ')');
        if (!res.ok) erros.push('Layout P: ' + res.mensagem);
      } catch(eP) { erros.push('Layout P: ' + eP.message); }
    }
    if (erros.length) return { ok: false, mensagem: erros.join('\n'), pdfs };
    return { ok: true, gravados: listaG.length + listaP.length, pdfs };
  } catch(e) { return { ok: false, mensagem: 'Erro cartão insumo: ' + e.message }; }
}

// ── Helpers de impressão ──────────────────────────────────────────────────────
function _buscarImagemDriveBase64(ref) {
  try {
    if (!ref || !String(ref).trim()) return '';
    let id = String(ref).trim();
    // Extrai ID do formato thumbnail URL, lh3.googleusercontent.com/d/ID, ou /file/d/ID/view
    if (id.includes('id=')) {
      const m = id.match(/id=([\w-]+)/);
      if (m) id = m[1];
    } else if (id.includes('/d/')) {
      const m = id.match(/\/d\/([\w-]+)/);
      if (m) id = m[1];
    } else if (id.includes('drive.google.com') || id.includes('googleusercontent.com')) {
      const m = id.match(/[-\w]{25,}/);
      if (m) id = m[0];
    }
    const blob = DriveApp.getFileById(id).getBlob();
    const mime = blob.getContentType() || 'image/jpeg';
    const b64  = Utilities.base64Encode(blob.getBytes());
    return 'data:' + mime + ';base64,' + b64;
  } catch(e) {
    Logger.log('Erro _buscarImagemDriveBase64(' + ref + '): ' + e.message);
    return '';
  }
}
